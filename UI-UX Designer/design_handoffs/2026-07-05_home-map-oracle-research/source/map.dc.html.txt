class Component extends DCLogic {
  state = { mode: "researchers", selected: null };

  select(kind, idx) {
    this.setState({ mode: kind, selected: idx });
  }

  renderVals() {
    const mode = this.state.mode;
    const list = mode === "researchers" ? RESEARCHERS : COMPANIES;
    const sel = this.state.selected != null ? list[this.state.selected] : null;

    const researchersOn = mode === "researchers";

    const mapDots = list.map((r, i) => ({
      x: r.x, y: r.y, r: this.state.selected === i ? 2.2 : 1.5,
      fill: mode === "companies" ? "#C77E5F" : "#00736E",
      onSelect: () => this.select(mode, i),
    }));

    const rows = list.map((r, i) => ({
      name: r.name, org: r.org, loc: r.loc, score: r.score,
      onSelect: () => this.select(mode, i),
    }));

    return {
      isResearchers: researchersOn,
      isCompanies: !researchersOn,
      showResearchers: () => this.setState({ mode: "researchers", selected: null }),
      showCompanies: () => this.setState({ mode: "companies", selected: null }),
      listTitle: researchersOn ? "Experts in your topics" : "Companies to watch",
      listCount: list.length,
      mapDots,
      rows,
      noneSelected: sel == null,
      hasSelected: sel != null,
      sel: sel || {},
      clearSelection: () => this.setState({ selected: null }),
    };
  }
}

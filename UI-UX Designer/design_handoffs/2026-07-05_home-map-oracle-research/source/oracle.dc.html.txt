<script type="text/x-dc" data-dc-script>
const PAPERS = {
  1: { id: 1, label: "Murray et al. 2021 · Off-flavours in pea protein isolates", title: "Off-flavours in pea protein isolates: origin and mitigation", author: "Murray, K. et al.", year: 2021, journal: "J. Agric. Food Chem.", abstract: "Reviews the volatile and non-volatile compounds responsible for beany/grassy notes in commercial pea protein isolates, identifying hexanal and related lipid-oxidation aldehydes as the dominant impact compounds across processing routes.", tags: ["pea protein", "hexanal", "lipid oxidation"] },
  2: { id: 2, label: "Sohail et al. 2022 · Masking strategies review", title: "Masking strategies for plant-protein off-notes: a comparative review", author: "Sohail, R. et al.", year: 2022, journal: "Trends Food Sci. Technol.", abstract: "Compares physical (cyclodextrin encapsulation, protein binding) and perceptual (Maillard top-note competition) masking strategies for off-notes in plant-based matrices, with guidance on combining both for durable sensory results.", tags: ["masking", "cyclodextrin", "Maillard"] },
  3: { id: 3, label: "Cerny 2003 · Fermentation & aroma modulation", title: "Fermentation-driven modulation of meaty aroma precursors", author: "Cerny, E.", year: 2003, journal: "Food Chem.", abstract: "Demonstrates that koji and lactic fermentation pre-treatments substantially reduce hexanal and related aldehydes prior to thermal processing, and shows additive effects when stacked with bitter-blocking compounds.", tags: ["fermentation", "koji", "aldehydes"] },
};

class Component extends DCLogic {
  state = { question: "", stage: "empty", modalId: null };

  onInput = (e) => this.setState({ question: e.target.value });
  onKeyDown = (e) => { if (e.key === "Enter" && !e.shiftKey) { e.preventDefault(); this.ask(); } };

  ask(text) {
    if (text) this.setState({ question: text });
    this.setState({ stage: "loading" });
    clearTimeout(this._t);
    this._t = setTimeout(() => this.setState({ stage: "answer" }), 1100);
  }

  openModal(id) {
    this.setState({ modalId: id });
  }

  renderVals() {
    const q = this.state.question || "";
    const modalPaper = this.state.modalId != null ? PAPERS[this.state.modalId] : null;

    return {
      question: q,
      onInput: this.onInput,
      onKeyDown: this.onKeyDown,
      onAsk: () => this.ask(),
      askDisabled: q.trim().length === 0,
      isEmpty: this.state.stage === "empty",
      isLoading: this.state.stage === "loading",
      isError: this.state.stage === "error",
      isAnswer: this.state.stage === "answer",
      loadingLabel: "Retrieving passages & writing a cited answer…",
      errorMessage: "The Oracle couldn't reach the literature database. Check your connection and try again.",
      sources: Object.values(PAPERS).map((p) => ({ id: p.id, label: p.label, onOpen: () => this.openModal(p.id) })),
      openCite1: () => this.openModal(1),
      openCite2: () => this.openModal(2),
      openCite3: () => this.openModal(3),
      pick0: () => this.ask("How do I mask beany off-notes in pea protein?"),
      pick1: () => this.ask("What are the key Maillard top-notes for plant-based beef?"),
      pick2: () => this.ask("Which volatiles drive the popcorn aroma of 2-acetyl-1-pyrroline?"),
      pick3: () => this.ask("What does fermentation contribute to meaty aroma profiles?"),
      pickF0: () => this.ask("Which cyclodextrin binds hexanal best?"),
      pickF1: () => this.ask("How much does koji lower hexanal?"),
      modalOpen: modalPaper != null,
      modalPaper: modalPaper || {},
      closeModal: () => this.setState({ modalId: null }),
      stopProp: (e) => e.stopPropagation(),
    };
  }
}

</script>
